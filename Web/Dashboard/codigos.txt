import threading
import time
import random

def start_websocket_test(socketio):

    def send_data():
        while True:
            valor = random.randint(350, 480)
            socketio.emit("co2", valor)
            time.sleep(2)

    thread = threading.Thread(target=send_data)
    thread.daemon = True
    thread.start()
